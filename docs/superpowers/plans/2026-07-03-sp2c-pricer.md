# SP2c — Pricer (joint matrix → fair contract prices) — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Turn the SP2b goal model's **joint score matrix** `P(home_goals, away_goals)` into a **fair probability** for every Kalshi soccer contract type, and price **same-game combos** by summing over the joint matrix (not multiplying marginals). This is design §4 component 4 (`pricer`).

**Architecture:** New `kalshi_edge/pricer/` subpackage. `contracts.py` is pure math over a joint matrix — each supported contract type is a predicate `(home_goals, away_goals) -> bool`, and its fair price is the sum of matrix cells where the predicate holds. `pricer.py` wraps a matrix in a `Pricer` with named methods (moneyline, totals, BTTS, team totals, exact score, double chance, clean sheet) and a `from_model` constructor, plus combo pricing: same-game legs intersect on one matrix; independent games multiply. Player-props pricing (per-player scoring rate) is **deferred to SP2d enrichment** — it needs the goalscorer/goals-per-90 data, not just the matrix.

**Tech Stack:** Python 3.11+, stdlib only. Reuses SP2b's `kalshi_edge/model/` (`score_matrix`, `outcome_probs`, `GoalModel`) and the SP1 pytest config.

**Key idea:** Every regulation-time contract is a subset of score outcomes. If `M[x][y] = P(home=x, away=y)`, then `fair = Σ_{(x,y): predicate(x,y)} M[x][y]`. Totals lines are half-integers (2.5, 3.5): `over(2.5)` = total ≥ 3; `under(2.5)` = total ≤ 2. A same-game combo predicate is the logical AND of its legs' predicates over the same matrix, so correlation is priced exactly.

# src/kalshi_edge/pricer/contracts.py
"""Kalshi soccer contract types as predicates over a joint score matrix.

A contract's fair probability is the sum of joint-matrix cells (x=home goals,
y=away goals) where its predicate holds. Totals lines are half-integers (2.5,
3.5, ...) so no push is possible. Regulation time only (matches the model).
"""
from collections.abc import Callable

Predicate = Callable[[int, int], bool]


def price_predicate(matrix: list[list[float]], predicate: Predicate) -> float:
    """Sum matrix cells where `predicate(home_goals, away_goals)` is True."""
    return sum(
        p
        for x, row in enumerate(matrix)
        for y, p in enumerate(row)
        if predicate(x, y)
    )


# --- moneyline (regulation) ---
def home_win(x: int, y: int) -> bool:
    return x > y


def draw(x: int, y: int) -> bool:
    return x == y


def away_win(x: int, y: int) -> bool:
    return x < y


# --- double chance ---
def home_or_draw(x: int, y: int) -> bool:
    return x >= y


def away_or_draw(x: int, y: int) -> bool:
    return x <= y


def not_draw(x: int, y: int) -> bool:
    return x != y


# --- totals (match total goals) ---
def over(line: float) -> Predicate:
    """P(total goals > line); line is a half-integer, e.g. over(2.5) => total >= 3."""
    return lambda x, y: (x + y) > line


def under(line: float) -> Predicate:
    return lambda x, y: (x + y) < line


# --- both teams to score ---
def both_teams_score(x: int, y: int) -> bool:
    return x >= 1 and y >= 1


def both_teams_score_no(x: int, y: int) -> bool:
    return x == 0 or y == 0


# --- team totals ---
def _goals(side: str, x: int, y: int) -> int:
    if side == "home":
        return x
    if side == "away":
        return y
    raise ValueError(f"side must be 'home' or 'away', got {side!r}")


def team_over(side: str, line: float) -> Predicate:
    _goals(side, 0, 0)  # validate side eagerly
    return lambda x, y: _goals(side, x, y) > line


def team_under(side: str, line: float) -> Predicate:
    _goals(side, 0, 0)
    return lambda x, y: _goals(side, x, y) < line


# --- clean sheet (named side concedes zero) ---
def clean_sheet(side: str) -> Predicate:
    _goals(side, 0, 0)
    other = "away" if side == "home" else "home"
    return lambda x, y: _goals(other, x, y) == 0


# --- exact score ---
def exact_score(home_goals: int, away_goals: int) -> Predicate:
    return lambda x, y: x == home_goals and y == away_goals
```

from kalshi_edge.data.cache import connect
from kalshi_edge.data.intl_results import load_results_csv
from kalshi_edge.model.goal_model import GoalModel
from kalshi_edge.pricer.pricer import Pricer

    f"2025-0{i%9+1}-15,Weak,Strong,0,3,Friendly,X,Y,TRUE\n" for i in range(10))


def _model():
    conn = connect(); load_results_csv(conn, ROWS)
    return GoalModel.from_db(conn, ref_date="2025-10-01")


def _pricer(home="Strong", away="Weak"):
    return Pricer.from_model(_model(), home, away, neutral=True)


def test_moneyline_matches_model_outcome_probs():
    m = _model()
    p = _pricer()
    ml = p.moneyline()
    ref = m.match_probabilities("Strong", "Weak", neutral=True)
    assert abs(ml["home"] - ref["home"]) < 1e-9
    assert abs(ml["draw"] - ref["draw"]) < 1e-9
    assert abs(ml["away"] - ref["away"]) < 1e-9
    assert round(sum(ml.values()), 6) == 1.0


def test_totals_complementary_and_named_helpers():
    p = _pricer()
    assert round(p.total_over(2.5) + p.total_under(2.5), 6) == 1.0
    assert 0.0 <= p.both_teams_score() <= 1.0


def test_price_named_dispatch_matches_methods():
    p = _pricer()
    assert abs(p.price("over", line=2.5) - p.total_over(2.5)) < 1e-12
    assert abs(p.price("home") - p.moneyline()["home"]) < 1e-12
    assert abs(p.price("btts") - p.both_teams_score()) < 1e-12


def test_same_game_combo_is_intersection_not_product():
    # Strong to win AND over 1.5 in the SAME game: must be the joint sum,
    # which for a strong favorite is well above the naive product only if
    # correlated — assert it equals the direct predicate intersection.
    p = _pricer()
    combo = p.same_game_combo([("home",), ("over", 2.5)])
    direct = contracts.price_predicate(
        p.matrix, lambda x, y: contracts.home_win(x, y) and contracts.over(2.5)(x, y))
    assert abs(combo - direct) < 1e-12
    # And it is NOT the product of the two marginals (correlation matters).
    product = p.price("home") * p.total_over(2.5)
    assert abs(combo - product) > 1e-6


def test_cross_game_combo_multiplies_independent_pricers():
    p1 = _pricer("Strong", "Weak")
    p2 = _pricer("Strong", "Weak")
    joint = Pricer.combo_across_games([
        (p1, [("home",)]),
        (p2, [("over", 2.5)]),
    ])
    expected = p1.price("home") * p2.total_over(2.5)
    assert abs(joint - expected) < 1e-12

# src/kalshi_edge/pricer/pricer.py
"""Fair prices for Kalshi soccer contracts from a joint score matrix.

`Pricer` wraps one match's joint matrix (from a `GoalModel`). Named methods
price the common contract types; `price(name, **kwargs)` dispatches by string.
Same-game combos are summed over the shared matrix (correlation priced exactly);
cross-game combos multiply the per-game leg probabilities (independence).
"""
from collections.abc import Sequence

from kalshi_edge.model.goal_model import GoalModel
from kalshi_edge.pricer import contracts

# Legs are (name,) or (name, line): "home"/"draw"/"away", "home_or_draw",
# "away_or_draw", "not_draw", "over"/"under" (needs line), "btts"/"btts_no",
# "home_over"/"home_under"/"away_over"/"away_under" (needs line),
# "home_clean_sheet"/"away_clean_sheet", "exact" (needs (h, a) as line).
Leg = tuple


def _predicate_for(name: str, line=None):
    simple = {
        "home": contracts.home_win,
        "draw": contracts.draw,
        "away": contracts.away_win,
        "home_or_draw": contracts.home_or_draw,
        "away_or_draw": contracts.away_or_draw,
        "not_draw": contracts.not_draw,
        "btts": contracts.both_teams_score,
        "btts_no": contracts.both_teams_score_no,
        "home_clean_sheet": contracts.clean_sheet("home"),
        "away_clean_sheet": contracts.clean_sheet("away"),
    }
    if name in simple:
        return simple[name]
    if name == "over":
        return contracts.over(line)
    if name == "under":
        return contracts.under(line)
    if name == "home_over":
        return contracts.team_over("home", line)
    if name == "home_under":
        return contracts.team_under("home", line)
    if name == "away_over":
        return contracts.team_over("away", line)
    if name == "away_under":
        return contracts.team_under("away", line)
    if name == "exact":
        return contracts.exact_score(line[0], line[1])
    raise ValueError(f"unknown contract: {name!r}")


class Pricer:
    def __init__(self, matrix: list[list[float]], home: str = "", away: str = ""):
        self.matrix = matrix
        self.home = home
        self.away = away

    @classmethod
    def from_model(cls, model: GoalModel, home: str, away: str,
                   neutral: bool = True) -> "Pricer":
        return cls(model.joint_matrix(home, away, neutral), home, away)

    # --- single contracts ---
    def price(self, name: str, line=None) -> float:
        return contracts.price_predicate(self.matrix, _predicate_for(name, line))

    def moneyline(self) -> dict:
        return {
            "home": self.price("home"),
            "draw": self.price("draw"),
            "away": self.price("away"),
        }

    def total_over(self, line: float) -> float:
        return self.price("over", line)

    def total_under(self, line: float) -> float:
        return self.price("under", line)

    def both_teams_score(self) -> float:
        return self.price("btts")

    # --- combos ---
    def _combo_predicate(self, legs: Sequence[Leg]):
        preds = [_predicate_for(leg[0], leg[1] if len(leg) > 1 else None) for leg in legs]
        return lambda x, y: all(p(x, y) for p in preds)

    def same_game_combo(self, legs: Sequence[Leg]) -> float:
        """P(all legs hit) within one match — summed over the shared matrix."""
        return contracts.price_predicate(self.matrix, self._combo_predicate(legs))

    @staticmethod
    def combo_across_games(games: Sequence[tuple["Pricer", Sequence[Leg]]]) -> float:
        """P(all legs hit) across independent matches — product of per-game combos."""
        prob = 1.0
        for pricer, legs in games:
            prob *= pricer.same_game_combo(legs)
        return prob
```
