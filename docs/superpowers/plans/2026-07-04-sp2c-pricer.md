# SP2c — Contract Pricer — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Turn the `GoalModel` joint score matrix into fair probabilities for every liquid Kalshi World Cup contract type — moneyline, totals, team totals, BTTS, correct score, and spread/handicap — so we can compare model fair value against Kalshi's de-vigged prices.

**Architecture:** New `kalshi_edge/pricer/` package. `prices.py` holds pure functions that read a joint score matrix `matrix[home_goals][away_goals]` and return probabilities. `price_match.py` ties a `GoalModel` matchup to a structured dict of all fair prices. Player-goal props are deferred to SP2d (they need player-level rates + confirmed lineups). Model-agnostic: everything consumes the matrix.

**Tech Stack:** Python 3.11+, stdlib only. Builds on SP2b (`model/`). SP1 pytest config.

**Matrix convention:** `matrix[x][y]` = P(home scores x, away scores y), from `GoalModel.joint_matrix(home, away, neutral)`. Lines like `2.5` mean the over/under boundary (over 2.5 ⇔ total ≥ 3).

**Kalshi contracts these map to:** KXWCGAME (moneyline), KXWCTOTAL (totals), KXWCTEAMTOTAL (team totals), KXWCBTTS (BTTS), KXWCSCORE (correct score), KXWCSPREAD (spread).

# src/kalshi_edge/pricer/prices.py
def _cells(matrix):
    for x, row in enumerate(matrix):
        for y, p in enumerate(row):
            yield x, y, p


def moneyline(matrix) -> dict:
    home = draw = away = 0.0
    for x, y, p in _cells(matrix):
        if x > y:
            home += p
        elif x == y:
            draw += p
        else:
            away += p
    return {"home": home, "draw": draw, "away": away}


def total_over(matrix, line: float) -> float:
    return sum(p for x, y, p in _cells(matrix) if x + y > line)


def total_under(matrix, line: float) -> float:
    return sum(p for x, y, p in _cells(matrix) if x + y < line)
```

# src/kalshi_edge/pricer/prices.py (append)
def team_total_over(matrix, line: float, side: str) -> float:
    if side == "home":
        return sum(p for x, y, p in _cells(matrix) if x > line)
    if side == "away":
        return sum(p for x, y, p in _cells(matrix) if y > line)
    raise ValueError(f"side must be 'home' or 'away', got {side!r}")


def btts(matrix) -> float:
    return sum(p for x, y, p in _cells(matrix) if x >= 1 and y >= 1)


def correct_score(matrix, home_goals: int, away_goals: int) -> float:
    if 0 <= home_goals < len(matrix) and 0 <= away_goals < len(matrix[0]):
        return matrix[home_goals][away_goals]
    return 0.0
```

# src/kalshi_edge/pricer/prices.py (append)
def spread_cover(matrix, line: float, side: str) -> float:
    """P(side wins by more than `line` goals). line 1.5 -> margin >= 2."""
    if side == "home":
        return sum(p for x, y, p in _cells(matrix) if (x - y) > line)
    if side == "away":
        return sum(p for x, y, p in _cells(matrix) if (y - x) > line)
    raise ValueError(f"side must be 'home' or 'away', got {side!r}")
```

# src/kalshi_edge/pricer/price_match.py
from kalshi_edge.pricer.prices import (
    moneyline, total_over, total_under, team_total_over, btts, spread_cover,
)

DEFAULT_TOTAL_LINES = [0.5, 1.5, 2.5, 3.5, 4.5, 5.5]
DEFAULT_TEAM_TOTAL_LINES = [0.5, 1.5, 2.5]
DEFAULT_SPREAD_LINES = [1.5, 2.5]


def price_match(model, home: str, away: str, neutral: bool = True,
                total_lines: list[float] | None = None,
                team_total_lines: list[float] | None = None,
                spread_lines: list[float] | None = None) -> dict:
    """Fair probabilities for all supported Kalshi contract types on this matchup."""
    total_lines = DEFAULT_TOTAL_LINES if total_lines is None else total_lines
    team_total_lines = DEFAULT_TEAM_TOTAL_LINES if team_total_lines is None else team_total_lines
    spread_lines = DEFAULT_SPREAD_LINES if spread_lines is None else spread_lines

    m = model.joint_matrix(home, away, neutral)
    return {
        "moneyline": moneyline(m),
        "totals": {ln: {"over": total_over(m, ln), "under": total_under(m, ln)}
                   for ln in total_lines},
        "team_totals": {
            "home": {ln: team_total_over(m, ln, "home") for ln in team_total_lines},
            "away": {ln: team_total_over(m, ln, "away") for ln in team_total_lines},
        },
        "btts": btts(m),
        "spreads": {
            "home": {ln: spread_cover(m, ln, "home") for ln in spread_lines},
            "away": {ln: spread_cover(m, ln, "away") for ln in spread_lines},
        },
    }
```

import os
import pytest
from kalshi_edge.data.cache import connect
from kalshi_edge.model.goal_model import GoalModel
from kalshi_edge.pricer.price_match import price_match

DB = "data/football.sqlite"


@pytest.mark.live
def test_price_match_on_real_matchup():
    if not os.path.exists(DB):
        pytest.skip("run scripts/build_db.py first")
    model = GoalModel.from_db(connect(DB), ref_date="2026-07-03", since="2016-01-01")
    prices = price_match(model, "Egypt", "Australia", neutral=True)
    ml = prices["moneyline"]
    assert round(ml["home"] + ml["draw"] + ml["away"], 6) == 1.0
    # Totals must be monotone: P(over 0.5) >= P(over 1.5) >= P(over 2.5)
    overs = [prices["totals"][ln]["over"] for ln in (0.5, 1.5, 2.5)]
    assert overs[0] >= overs[1] >= overs[2]
